**Message send + streaming read:**
```typescript
async function handleSubmit() {
  const userMsg = { role: "user" as const, content: input.trim() };
  const newMessages = [...messages, userMsg];
  setMessages(newMessages);
  setInput("");
  setIsLoading(true);

  try {
    const res = await fetch("/api/admin-chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ messages: newMessages, context }),
    });

    if (!res.ok) {
      const err = await res.json().catch(() => ({ error: "Unknown error" }));
      setMessages(prev => [...prev, { role: "assistant", content: `Error: ${err.error}` }]);
      return;
    }

    // Read streaming response
    const reader = res.body!.getReader();
    const decoder = new TextDecoder();
    let assistantContent = "";

    // Add empty assistant message that we'll update
    setMessages(prev => [...prev, { role: "assistant", content: "" }]);

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      
      // Decode the SSE chunk and extract text
      const chunk = decoder.decode(value, { stream: true });
      // Parse SSE events — each event has: event: content_block_delta, data: {...}
      // The Anthropic SDK's toReadableStream() sends raw SSE events
      // Extract text from content_block_delta events
      const lines = chunk.split("\n");
      for (const line of lines) {
        if (line.startsWith("data: ")) {
          try {
            const data = JSON.parse(line.slice(6));
            if (data.type === "content_block_delta" && data.delta?.text) {
              assistantContent += data.delta.text;
              setMessages(prev => {
                const updated = [...prev];
                updated[updated.length - 1] = { role: "assistant", content: assistantContent };
                return updated;
              });
            }
          } catch { /* skip non-JSON lines */ }
        }
      }
    }
  } catch {
    setMessages(prev => [...prev, { role: "assistant", content: "Error: Failed to connect. Please try again." }]);
  } finally {
    setIsLoading(false);
  }
}
```

**Important notes:**
- The Anthropic SDK's `stream.toReadableStream()` returns raw SSE format. The client needs to parse `data:` lines and extract `content_block_delta` events. If the SSE parsing above doesn't work correctly with the SDK version, fall back to reading raw text chunks — the SDK may also support a simpler byte stream. Test and adjust. Add a code comment documenting the expected SSE format.
- `z-40` ensures the widget stays below the Sheet sidebar (z-50) used for mobile navigation
- Session-only history: messages reset when page navigates away (component unmounts). This is intentional — no persistence needed per research decision.
- Do NOT use `dangerouslySetInnerHTML` — render message content as plain text with `whitespace-pre-wrap`
